import argparse
import datetime
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import time

parser = argparse.ArgumentParser()
parser.add_argument('--repo', required=True)
parser.add_argument('--output', required=True)
parser.add_argument('--name', required=True)
parser.add_argument('--environment')
parser.add_argument('command', nargs=argparse.REMAINDER)
arguments = parser.parse_args()
repository = Path(arguments.repo).resolve()
output = Path(arguments.output).resolve()
output.mkdir(parents=True, exist_ok=True)
log = output / (arguments.name + '.log')
journal = output / (arguments.name + '-execution.json')
if log.exists() or journal.exists():
    raise SystemExit('Refusing to overwrite an existing capture')

def identity():
    def git(*command):
        return subprocess.check_output(['git', *command], cwd=repository, text=True).strip()
    return {'commit': git('rev-parse', 'HEAD'), 'tree': git('rev-parse', 'HEAD^{tree}'),
            'status': git('status', '--porcelain=v1', '--untracked-files=all')}

environment = os.environ.copy()
resources = {'SHARPFORGE_MAX_PARALLEL_RUNS': '1', 'SHARPFORGE_TEST_CONCURRENCY': '1',
             'SHARPFORGE_MAX_OLD_SPACE_MB': '512'}
environment.update(resources)
if arguments.environment:
    environment.update(json.loads(Path(arguments.environment).read_text()))
command = arguments.command
if command and command[0] == '--':
    command = command[1:]
node = shutil.which('node')
resolved = [node if token == '{node}' else token for token in command]
if not resolved:
    raise SystemExit('A command is required')
before = identity()
if before['status']:
    raise SystemExit('Capture requires a clean worktree; first paths: ' + before['status'][:1500])
record = {'format': 'SharpForge.A05CommandCapture/1', 'name': arguments.name,
          'command': resolved, 'cwd': str(repository),
          'resources': {**resources, 'NODE_OPTIONS': environment.get('NODE_OPTIONS')},
          'toolchainEnvironment': {key: environment.get(key) for key in
              ['DOTNET', 'DOTNET_ROOT', 'DOTNET_SDK_VERSION', 'DOTNET_TARGET_FRAMEWORK', 'SHARPFORGE_REFERENCE_PACK']},
          'node': subprocess.check_output([node, '--version'], text=True).strip(),
          'startIdentity': before,
          'startedUTC': datetime.datetime.now(datetime.timezone.utc).isoformat()}
journal.write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps({'started': arguments.name, 'commit': before['commit'], 'command': resolved}), flush=True)
started = time.monotonic()
with log.open('xb') as stream:
    process = subprocess.run(resolved, cwd=repository, env=environment, stdout=stream, stderr=subprocess.STDOUT)
record.update({'completedUTC': datetime.datetime.now(datetime.timezone.utc).isoformat(),
               'elapsedSeconds': time.monotonic() - started, 'exitCode': process.returncode,
               'endIdentity': identity()})
content = log.read_bytes()
record['artifact'] = {'path': log.name, 'bytes': len(content), 'sha256': hashlib.sha256(content).hexdigest()}
journal.write_text(json.dumps(record, indent=2) + '\n')
print(json.dumps({key: record[key] for key in ['name', 'exitCode', 'elapsedSeconds', 'endIdentity', 'artifact']}), flush=True)
print(content.decode('utf-8', errors='replace')[-5000:], flush=True)
raise SystemExit(process.returncode if process.returncode >= 0 else 128 - process.returncode)
